// --- НАСТРОЙКИ ---
#define PL_BLOCK_SIZE 188
#define MAGIC_H 0xFA
#define MAGIC_L 0xCE
#define RING_SIZE (PL_BLOCK_SIZE * 64)

extern XLlFifo fifo;

static uint8_t tx_ring[RING_SIZE];
static uint16_t tx_head = 0, tx_tail = 0, tx_count = 0;

static uint8_t rx_ring[RING_SIZE];
static uint16_t rx_head = 0, rx_tail = 0, rx_count = 0;

// --- ОТЛАДОЧНЫЕ СЧЕТЧИКИ ---
static u32 tx_dbg_pkts_in = 0;    // Сколько пакетов (IP/ARP) пришло от lwIP
static u32 tx_dbg_blocks_out = 0; // Сколько блоков 188 ушло в FPGA
static u32 rx_dbg_blocks_in = 0;  // Сколько блоков 188 пришло из FPGA
static u32 rx_dbg_pkts_out = 0;   // Сколько пакетов собрали и отдали в lwIP
static u32 rx_dbg_err_magic = 0;  // Ошибки синхронизации (потерянные заголовки)

typedef enum { ST_MAGIC_H, ST_MAGIC_L, ST_LEN_H, ST_LEN_L, ST_DATA } rx_state_t;
static rx_state_t rx_state = ST_MAGIC_H;
static uint16_t rx_pkt_len = 0;
static uint16_t rx_pkt_received = 0;
static uint8_t rx_pkt_buffer[1600];

static void send_block_to_pl(uint8_t *block) {
    // ВРЕМЕННЫЙ ТЕСТ: Игнорируем переполнение FIFO
    // Просто считаем, что отправили, даже если FIFO полон

    if (XLlFifo_iTxVacancy(&fifo) >= (PL_BLOCK_SIZE / 4)) {
        u32 word;
        for (int i = 0; i < PL_BLOCK_SIZE; i += 4) {
            word =  (u32)block[i]        | ((u32)block[i+1] << 8) |
                   ((u32)block[i+2] << 16)| ((u32)block[i+3] << 24);
            XLlFifo_TxPutWord(&fifo, word);
        }
        XLlFifo_iTxSetLen(&fifo, PL_BLOCK_SIZE);
    }
    else {
        // FIFO полон!
        // Обычно мы тут ничего не делаем, но для теста сбросим его,
        // чтобы увидеть, что счетчики растут.
        XLlFifo_Reset(&fifo);
    }

    tx_dbg_blocks_out++; // Всегда увеличиваем счетчик для визуализации
}

// ================= TX =================
static void tx_push(uint8_t b) {
    tx_ring[tx_head] = b;
    tx_head = (tx_head + 1) % RING_SIZE;
    tx_count++;
}

static void tx_process() {
    while (tx_count >= PL_BLOCK_SIZE) {
        uint8_t block[PL_BLOCK_SIZE];
        for (int i = 0; i < PL_BLOCK_SIZE; i++) {
            block[i] = tx_ring[tx_tail];
            tx_tail = (tx_tail + 1) % RING_SIZE;
        }
        tx_count -= PL_BLOCK_SIZE;
        send_block_to_pl(block);
    }
}

err_t radio_linkoutput(struct netif *netif, struct pbuf *p) {
    if ((RING_SIZE - tx_count) < (p->tot_len + 4)) return ERR_MEM;

    tx_dbg_pkts_in++; // +1 пакет зашел на нарезку

    // Заголовок
    tx_push(MAGIC_H); tx_push(MAGIC_L);
    tx_push((p->tot_len >> 8) & 0xFF);
    tx_push((p->tot_len & 0xFF));

    //TODO: cnt pkt, cnt seg, crc at the end of pkt.

    // Данные
    struct pbuf *q;
    for(q = p; q != NULL; q = q->next) {
        uint8_t *data = (uint8_t*)q->payload;
        for(int i = 0; i < q->len; i++) tx_push(data[i]);
    }
    tx_process();
    return ERR_OK;
}

void radio_tx_periodic_flush(void) {
    if (tx_count > 0 && tx_count < PL_BLOCK_SIZE) {
        while (tx_count < PL_BLOCK_SIZE) tx_push(0x00);
        tx_process();
    }
}

// ================= RX =================
void radio_input_block(uint8_t *block_188, struct netif *inp_netif) {
    rx_dbg_blocks_in++; // +1 блок пришел из радио

    for (int i = 0; i < PL_BLOCK_SIZE; i++) {
        rx_ring[rx_head] = block_188[i];
        rx_head = (rx_head + 1) % RING_SIZE;
        rx_count++;
    }

    while (rx_count > 0) {
        uint8_t b = rx_ring[rx_tail];

        switch (rx_state) {
            case ST_MAGIC_H:
                if (b == MAGIC_H) { rx_state = ST_MAGIC_L; }
                else { /* Ждем синхронизации */ }
                rx_tail = (rx_tail + 1) % RING_SIZE; rx_count--;
                break;

            case ST_MAGIC_L:
                if (b == MAGIC_L) { rx_state = ST_LEN_H; }
                else {
                    rx_state = ST_MAGIC_H;
                    rx_dbg_err_magic++; // Ошибка синхра
                }
                rx_tail = (rx_tail + 1) % RING_SIZE; rx_count--;
                break;

            case ST_LEN_H:
                rx_pkt_len = (b << 8);
                rx_state = ST_LEN_L;
                rx_tail = (rx_tail + 1) % RING_SIZE; rx_count--;
                break;

            case ST_LEN_L:
                rx_pkt_len |= b;
                rx_tail = (rx_tail + 1) % RING_SIZE; rx_count--;
                if (rx_pkt_len > 1520 || rx_pkt_len == 0) rx_state = ST_MAGIC_H;
                else { rx_pkt_received = 0; rx_state = ST_DATA; }
                break;

            case ST_DATA:
                rx_pkt_buffer[rx_pkt_received++] = b;
                rx_tail = (rx_tail + 1) % RING_SIZE; rx_count--;

                if (rx_pkt_received == rx_pkt_len) {
                    struct pbuf *p = pbuf_alloc(PBUF_RAW, rx_pkt_len, PBUF_POOL);
                    if (p) {
                        pbuf_take(p, rx_pkt_buffer, rx_pkt_len);
                        if (inp_netif->input(p, inp_netif) != ERR_OK) pbuf_free(p);
                        else rx_dbg_pkts_out++; // +1 пакет успешно собран
                    }
                    rx_state = ST_MAGIC_H;
                }
                break;
        }
    }
}

// ФУНКЦИЯ ВЫВОДА СТАТИСТИКИ
void radio_stream_print_stats(void) {
    xil_printf("[RF STREAM]: TX Pkts: %u -> Blks: %u | RX Blks: %u -> Pkts: %u (Err: %u)\r\n",
               tx_dbg_pkts_in, tx_dbg_blocks_out,
               rx_dbg_blocks_in, rx_dbg_pkts_out, rx_dbg_err_magic);
}

void radio_stream_init(void) {
    tx_head = tx_tail = tx_count = 0;
    rx_head = rx_tail = rx_count = 0;
    // Сброс счетчиков при старте
    tx_dbg_pkts_in = tx_dbg_blocks_out = 0;
    rx_dbg_blocks_in = rx_dbg_pkts_out = rx_dbg_err_magic = 0;
}
